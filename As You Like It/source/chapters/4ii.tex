\Scene{2}[The Forest of Arden]

\enter*{\character{Jaques} and \character{Lords,} like foresters}


\speech[Jaques]{Which is he that killed the deer?}

\speech[First Lord]{Sir, it was I.}

\speech[Jaques][Lords]{Let's present him to the Duke like a Roman conqueror. And it would do well to set the deer's horns upon his head for a branch of victory.—Have you no song, forester, for this purpose?}

\speech[Second Lord]{Yes, sir.}

\speech[Jaques]{Sing it. 'Tis no matter how it be in tune, so it make noise enough.}

\speech[Second Lord]{
	\stage{Sings}
	\indentpattern{001}
		\begin{verse}\itshape
			\begin{patverse}
				What shall he have that killed the deer?\\
				His leather skin and horns to wear.\\
				   Then sing him home.\\
				   \end{patverse}
		\end{verse}
}
