\Scene{2}[The Forest of Arden]

\enter{\textsc{Jaques} and \textsc{Lords,} like foresters}


\begin{prose_speech}[Jaques]  Which is he that killed the deer?
\end{prose_speech}

\begin{prose_speech}[First Lord]  Sir, it was I.
\end{prose_speech}

\begin{prose_speech}[Jaques]
	\stage{To the other \textsc{Lords}}  Let's present him to the Duke like a Roman conqueror. And it would do well to set the deer's horns upon his head for a branch of victory.—Have you no song, forester, for this purpose?
\end{prose_speech}

\begin{prose_speech}[Second Lord]  Yes, sir.
\end{prose_speech}

\begin{prose_speech}[Jaques]  Sing it. 'Tis no matter how it be in tune, so it make noise enough.
\end{prose_speech}

\begin{prose_speech}[Second Lord] 
	\stage{Sings}
		\indentpattern{001}
\vspace{-.5\baselineskip}
\begin{verse}\itshape
\begin{patverse}
	What shall he have that killed the deer?\\
	His leather skin and horns to wear.\\
	   Then sing him home.\\
	   \end{patverse}
\end{verse}
\end{prose_speech}
